cmake_minimum_required(VERSION 3.18)
project(aurora_fv LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CFD_EXTERNALS_ROOT "${CMAKE_SOURCE_DIR}/../external/cfd_externals/install" CACHE PATH "CGNS/METIS installation prefix")
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(OpenMP REQUIRED)
find_path(CGNS_INCLUDE_DIR cgnslib.h HINTS "${CFD_EXTERNALS_ROOT}/include" REQUIRED)
find_library(CGNS_LIBRARY cgns HINTS "${CFD_EXTERNALS_ROOT}/lib" REQUIRED)
find_path(METIS_INCLUDE_DIR metis.h HINTS "${CFD_EXTERNALS_ROOT}/include" REQUIRED)
find_library(METIS_LIBRARY metis HINTS "${CFD_EXTERNALS_ROOT}/lib" REQUIRED)
find_path(NLOHMANN_INCLUDE_DIR nlohmann/json.hpp HINTS "${CMAKE_SOURCE_DIR}/../external/nlohmann" REQUIRED)

add_executable(aurora-fv src/main.cpp)
target_include_directories(aurora-fv PRIVATE ${CGNS_INCLUDE_DIR} ${METIS_INCLUDE_DIR} ${NLOHMANN_INCLUDE_DIR})
target_link_libraries(aurora-fv PRIVATE MPI::MPI_CXX OpenMP::OpenMP_CXX ${CGNS_LIBRARY} ${METIS_LIBRARY})
target_compile_options(aurora-fv PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(aurora-fv PROPERTIES BUILD_RPATH "${CFD_EXTERNALS_ROOT}/lib")
