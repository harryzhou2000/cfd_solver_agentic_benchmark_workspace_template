cmake_minimum_required(VERSION 3.18)
project(cfd_mpi_solver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CFD_EXTERNALS_ROOT "$ENV{CFD_EXTERNALS_ROOT}" CACHE PATH "external dependency prefix")
if(NOT CFD_EXTERNALS_ROOT)
  set(CFD_EXTERNALS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../external/cfd_externals/install")
endif()
find_package(MPI REQUIRED COMPONENTS CXX)
find_path(CGNS_INCLUDE_DIR cgnslib.h HINTS "${CFD_EXTERNALS_ROOT}/include")
find_library(CGNS_LIBRARY cgns HINTS "${CFD_EXTERNALS_ROOT}/lib")
find_path(METIS_INCLUDE_DIR metis.h HINTS "${CFD_EXTERNALS_ROOT}/include")
find_library(METIS_LIBRARY metis HINTS "${CFD_EXTERNALS_ROOT}/lib")
if(NOT CGNS_INCLUDE_DIR OR NOT CGNS_LIBRARY OR NOT METIS_INCLUDE_DIR OR NOT METIS_LIBRARY)
 message(FATAL_ERROR "CGNS and METIS are required; set CFD_EXTERNALS_ROOT")
endif()
add_executable(cfd_solver src/main.cpp)
target_include_directories(cfd_solver PRIVATE ${CGNS_INCLUDE_DIR} ${METIS_INCLUDE_DIR})
target_link_libraries(cfd_solver PRIVATE MPI::MPI_CXX ${CGNS_LIBRARY} ${METIS_LIBRARY})
target_compile_options(cfd_solver PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(cfd_solver PROPERTIES BUILD_RPATH "${CFD_EXTERNALS_ROOT}/lib")

# Kept separate from the solver so new CGNS meshes can be audited without an
# MPI run.  This is particularly useful for checking multi-zone interfaces and
# the element-id ranges used by EdgeCenter boundary conditions.
add_executable(cgns_mesh_inspect tools/cgns_mesh_inspect.cpp)
target_include_directories(cgns_mesh_inspect PRIVATE ${CGNS_INCLUDE_DIR})
target_link_libraries(cgns_mesh_inspect PRIVATE ${CGNS_LIBRARY})
target_compile_options(cgns_mesh_inspect PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(cgns_mesh_inspect PROPERTIES BUILD_RPATH "${CFD_EXTERNALS_ROOT}/lib")
